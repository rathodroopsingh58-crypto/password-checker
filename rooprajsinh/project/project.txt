import tkinter as tk
from tkinter import messagebox
import re


# ---------------- PASSWORD CHECKER ---------------- #

def check_password():
    password = password_entry.get()

    if password == "":
        strength_label.config(text="Enter a password", fg="orange")
        progress_bar.config(width=0)
        return

    score = 0

    # 1. Check length
    if len(password) >= 8:
        score += 1
        length_label.config(text="? At least 8 characters", fg="lime green")
    else:
        length_label.config(text="? At least 8 characters", fg="red")

    # 2. Check uppercase
    if re.search(r"[A-Z]", password):
        score += 1
        upper_label.config(text="? Contains uppercase letter", fg="lime green")
    else:
        upper_label.config(text="? Contains uppercase letter", fg="red")

    # 3. Check lowercase
    if re.search(r"[a-z]", password):
        score += 1
        lower_label.config(text="? Contains lowercase letter", fg="lime green")
    else:
        lower_label.config(text="? Contains lowercase letter", fg="red")

    # 4. Check number
    if re.search(r"[0-9]", password):
        score += 1
        number_label.config(text="? Contains a number", fg="lime green")
    else:
        number_label.config(text="? Contains a number", fg="red")

    # 5. Check special character
    if re.search(r"[^A-Za-z0-9]", password):
        score += 1
        special_label.config(
            text="? Contains special character", fg="lime green"
        )
    else:
        special_label.config(
            text="? Contains special character", fg="red"
        )

    # ---------------- STRENGTH ---------------- #

    if score <= 1:
        strength = "Very Weak"
        color = "red"
        bar_width = 80

    elif score == 2:
        strength = "Weak"
        color = "red"
        bar_width = 150

    elif score == 3:
        strength = "Medium"
        color = "orange"
        bar_width = 230

    elif score == 4:
        strength = "Strong"
        color = "yellow"
        bar_width = 310

    else:
        strength = "Very Strong"
        color = "lime green"
        bar_width = 390

    strength_label.config(text=strength, fg=color)
    score_label.config(text=f"Score: {score}/5")

    progress_bar.config(
        width=bar_width,
        bg=color
    )


# ---------------- SHOW / HIDE PASSWORD ---------------- #

def show_password():
    if password_entry.cget("show") == "*":
        password_entry.config(show="")
        show_button.config(text="Hide")
    else:
        password_entry.config(show="*")
        show_button.config(text="Show")


# ---------------- CLEAR ---------------- #

def clear_password():
    password_entry.delete(0, tk.END)

    strength_label.config(
        text="Password Strength",
        fg="white"
    )

    score_label.config(text="Score: 0/5")

    progress_bar.config(width=0)

    length_label.config(
        text="? At least 8 characters",
        fg="lightgray"
    )

    upper_label.config(
        text="? Contains uppercase letter",
        fg="lightgray"
    )

    lower_label.config(
        text="? Contains lowercase letter",
        fg="lightgray"
    )

    number_label.config(
        text="? Contains a number",
        fg="lightgray"
    )

    special_label.config(
        text="? Contains special character",
        fg="lightgray"
    )
